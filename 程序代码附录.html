贝叶斯近似测量不变性模拟研究程序代码

1 研究设计与模型说明

本附录给出蒙特卡罗模拟的数据生成、模型设定、模型拟合、收敛诊断、参数返真性、非不变检测以及结果汇总与作图代码。

主模拟固定为10个组、6个连续观测指标。非不变结构共7种：完全不变1种，以及由非不变幅度（0.1、0.2）与非不变题目数（1、2、3题）交叉形成的6种条件。每组样本量设置为100、500和1000。因此，7种非不变结构与3种样本量组合形成21种模拟条件，每个条件重复500次。

比较的模型包括三种：小方差正态先验模型（small-variance normal prior，SVN）将组间参数差异设为均值0、方差0.01的正态先验；弱稀疏 spike-and-slab 模型（SSP-W）对非零成分的包含概率设定 π ~ Beta(1, 1)，其先验均值为0.50，对稀疏程度不作方向性偏好；强稀疏 spike-and-slab 模型（SSP-S）设定 π ~ Beta(1, 4)，其先验均值为0.20，更倾向于认为多数参数保持不变。SSP-W与SSP-S的模型结构相同，仅包含概率先验不同，分别体现较弱与较强的稀疏性假设。

为使模型可识别，所有模型均以最后一组作为参考组；在主模拟中，将第10组的因子均值和因子方差分别固定为0和1，并将每一组第1题的因子载荷固定为1。

3.2 固定的非不变位置和偏移方向

测量非不变通过对选定题项的因子载荷和截距施加 ±0.1 或 ±0.2 的扰动实现，分别表示小幅与中等偏离。这一效应量设定与近似测量不变性及项目功能差异研究中的常见做法一致（Muthén & Asparouhov, 2012；Brandt et al., 2025）。

为保证不同模拟条件之间的可比性，非不变参数的位置采用预先设定的结构化分配方式。随着非不变题目数增加，新增位置在前一条件的基础上递增扩展，而非重新随机抽取。载荷与截距的非不变参数分布在不同题项与组别上，两类参数的总体非不变比例保持一致；主模拟中，同一组同一题项不同时设定为载荷与截距非不变。为保持测量结构稳定性，第1、6和第10组的全部测量参数，以及第1题和第4题的测量参数在所有条件下保持不变。具体位置和偏移方向见下列矩阵：0表示参数不变，1表示正向偏移，-1表示负向偏移。

# 将方向矩阵乘以偏移幅度，并叠加到完全不变的基准参数上
make_main_truth <- function(pattern, magnitude) {
  direction <- direction_patterns[[pattern]]

  list(
    lambda = lambda_invariant + magnitude * direction$lambda,
    nu = nu_invariant + magnitude * direction$nu,
    alpha_f = alpha_10,
    psi = psi_10,
    theta = theta_6,
    lambda_direction = direction$lambda,
    nu_direction = direction$nu,
    magnitude = magnitude,
    pattern = pattern
  )
}

3.5 生成一个条件的500份数据

# 一个条件只指定样本量、非不变结构与偏移幅度；位置由上节矩阵读取
generate_condition <- function(condition, output_root, n_rep = 500L,
                               master_seed = 123L) {
  true_par <- make_main_truth(
    pattern = condition$pattern,
    magnitude = condition$magnitude
  )

  G <- condition$G
  P <- condition$P
  N_g <- condition$N_g
  N <- G * N_g

  condition_dir <- file.path(
    output_root,
    paste0("Non_", format(condition$magnitude, nsmall = 1)),
    paste0("N", N_g),
    condition$pattern
  )
  data_dir <- file.path(condition_dir, "data")
  dir.create(data_dir, recursive = TRUE, showWarnings = FALSE)

  set.seed(master_seed)
  seed_vec <- sample.int(1e7, size = n_rep, replace = FALSE)
  saveRDS(seed_vec, file.path(condition_dir, "seed_list.rds"))

  for (i in seq_len(n_rep)) {
    set.seed(seed_vec[i])
    group <- rep(seq_len(G), each = N_g)
    Y <- matrix(NA_real_, nrow = N, ncol = P)

    for (g in seq_len(G)) {
      index <- which(group == g)
      factor_score <- rnorm(
        N_g,
        mean = true_par$alpha_f[g],
        sd = sqrt(true_par$psi[g])
      )

      Y[index, ] <-
        matrix(true_par$nu[g, ], N_g, P, byrow = TRUE) +
        factor_score %o% true_par$lambda[g, ] +
        matrix(
          rnorm(N_g * P, 0, sqrt(true_par$theta)),
          nrow = N_g,
          ncol = P
        )
    }

    data_object <- list(
      Y = Y,
      group = group,
      G = G,
      P = P,
      N_g = N_g,
      N = N,
      Y_means = colMeans(Y),
      Y_sds = apply(Y, 2, sd),
      true_par = true_par,
      condition = condition,
      replication = i,
      seed = seed_vec[i]
    )

    saveRDS(
      data_object,
      file.path(data_dir, paste0("data_sim_", i, ".rds"))
    )
  }

  invisible(condition_dir)
}

4 JAGS模型

4.1 SVN模型

model {
  for (i in 1:N) {
    f[i] ~ dnorm(alpha_f[group[i]], prec_f[group[i]])
    for (p in 1:P) {
      Y[i,p] ~ dnorm(
        nu[group[i],p] + lambda[group[i],p] * f[i],
        prec_theta[p]
      )
    }
  }

  for (g in 1:(G-1)) {
    alpha_f[g] ~ dnorm(0, 1 / pow(1.0, 2))
    prec_f[g] ~ dgamma(1, 1)
    sigma_f[g] <- 1 / sqrt(prec_f[g])
  }

  alpha_f[G] <- 0
  sigma_f[G] <- 1
  prec_f[G] <- 1

  for (p in 1:P) {
    prec_theta[p] ~ dgamma(1, 1)
    theta[p] <- 1 / prec_theta[p]
  }

  lambda[G,1] <- 1
  nu[G,1] ~ dnorm(0, 1 / pow(1.0, 2))

  for (p in 2:P) {
    lambda[G,p] ~ dnorm(0, 1 / pow(1.0, 2))
    nu[G,p] ~ dnorm(0, 1 / pow(1.0, 2))
  }

  sigma_lambda <- 0.1
  sigma_nu <- 0.1

  for (g in 1:(G-1)) {
    lambda[g,1] <- 1

    z_nu[g,1] ~ dnorm(0, 1)
    delta_nu[g,1] <- sigma_nu * z_nu[g,1]
    nu[g,1] <- nu[G,1] + delta_nu[g,1]

    for (p in 2:P) {
      z_lambda[g,p] ~ dnorm(0, 1)
      delta_lambda[g,p] <- sigma_lambda * z_lambda[g,p]
      lambda[g,p] <- lambda[G,p] + delta_lambda[g,p]

      z_nu[g,p] ~ dnorm(0, 1)
      delta_nu[g,p] <- sigma_nu * z_nu[g,p]
      nu[g,p] <- nu[G,p] + delta_nu[g,p]
    }
  }

  for (g in 1:G) {
    psi[g] <- pow(sigma_f[g], 2)
  }
}

4.2 SSP模型

SSP-W与SSP-S共用下列模型结构，但包含概率的Beta先验不同。SSP-W设定 πλ、πν ~ Beta(1, 1)，对应代码中的 pi_a = 1, pi_b = 1；SSP-S设定 πλ、πν ~ Beta(1, 4)，对应 pi_a = 1, pi_b = 4。其中，πλ和πν分别表示载荷差异与截距差异属于非零成分的先验概率。

model {
  # 测量模型与潜变量分布
  for (i in 1:N) {
    f[i] ~ dnorm(alpha_f[group[i]], prec_f[group[i]])
    for (p in 1:P) {
      Y[i,p] ~ dnorm(
        nu[group[i],p] + lambda[group[i],p] * f[i],
        prec_theta[p]
      )
    }
  }

  for (g in 1:(G-1)) {
    alpha_f[g] ~ dnorm(0, 1 / pow(1.0, 2))
    sigma_f[g] ~ dt(0, 1 / pow(1.5, 2), 3) T(0,)
    prec_f[g] <- 1 / pow(sigma_f[g], 2)
  }

  # 最后一组为参考组
  alpha_f[G] <- 0
  sigma_f[G] <- 1
  prec_f[G] <- 1

  for (p in 1:P) {
    sigma_theta[p] ~ dt(0, 1 / pow(1, 2), 3) T(0,)
    prec_theta[p] <- 1 / pow(sigma_theta[p], 2)
    theta[p] <- pow(sigma_theta[p], 2)
  }

  lambda[G,1] <- 1
  nu[G,1] ~ dnorm(0, 1 / pow(1.0, 2))

  for (p in 2:P) {
    lambda[G,p] ~ dnorm(0, 1 / pow(1.0, 2))
    nu[G,p] ~ dnorm(0, 1 / pow(1.0, 2))
  }

  # spike-and-slab：gamma=0时差异为0，gamma=1时进入slab
  for (g in 1:(G-1)) {
    lambda[g,1] <- 1

    gamma_nu[g,1] ~ dbern(pi_nu)
    z_nu[g,1] ~ dnorm(0, 1)
    slab_delta_nu[g,1] <- z_nu[g,1] * slab_sd
    delta_nu[g,1] <- gamma_nu[g,1] * slab_delta_nu[g,1]
    nu[g,1] <- nu[G,1] + delta_nu[g,1]

    for (p in 2:P) {
      gamma_lambda[g,p] ~ dbern(pi_lambda)
      gamma_nu[g,p] ~ dbern(pi_nu)

      z_lambda[g,p] ~ dnorm(0, 1)
      z_nu[g,p] ~ dnorm(0, 1)

      slab_delta_lambda[g,p] <- z_lambda[g,p] * slab_sd
      slab_delta_nu[g,p] <- z_nu[g,p] * slab_sd

      delta_lambda[g,p] <-
        gamma_lambda[g,p] * slab_delta_lambda[g,p]
      delta_nu[g,p] <- gamma_nu[g,p] * slab_delta_nu[g,p]

      lambda[g,p] <- lambda[G,p] + delta_lambda[g,p]
      nu[g,p] <- nu[G,p] + delta_nu[g,p]
    }
  }

  # SSP-W传入(1,1)，SSP-S传入(1,4)
  pi_lambda ~ dbeta(pi_a, pi_b)
  pi_nu ~ dbeta(pi_a, pi_b)

  slab_sd ~ dt(0, 1 / pow(0.5, 2), 3) T(0,)

  for (g in 1:G) {
    psi[g] <- pow(sigma_f[g], 2)
  }
}

5 模型拟合

5.1 SSP初始值

SVN采用JAGS默认初始值。SSP包含离散选择指标，因此为各链设置不同的合理初始状态。

make_ssp_inits <- function(G, P, dataset_id, chain_id,
                           slab_start = 0.6,
                           pi_range = c(0.2, 0.5)) {
  set.seed(100000L + dataset_id * 10L + chain_id)

  pi_lambda <- runif(1, pi_range[1], pi_range[2])
  pi_nu <- runif(1, pi_range[1], pi_range[2])
  slab_sd <- exp(rnorm(1, log(slab_start), 0.25))

  gamma_lambda <- matrix(
    rbinom((G - 1L) * P, 1L, pi_lambda),
    nrow = G - 1L,
    ncol = P
  )
  gamma_lambda[, 1] <- 0L

  gamma_nu <- matrix(
    rbinom((G - 1L) * P, 1L, pi_nu),
    nrow = G - 1L,
    ncol = P
  )

  list(
    pi_lambda = pi_lambda,
    pi_nu = pi_nu,
    slab_sd = slab_sd,
    gamma_lambda = gamma_lambda,
    gamma_nu = gamma_nu,
    .RNG.name = "base::Mersenne-Twister",
    .RNG.seed = 200000L + dataset_id * 10L + chain_id
  )
}

5.2 拟合单个数据集

# method决定模型文件、结果目录以及SSP包含概率的Beta先验
fit_one_dataset <- function(data_file, method,
                            svn_model_file = "models/svn.jags",
                            ssp_model_file = "models/ssp.jags") {
  data_object <- readRDS(data_file)

  data_jags <- list(
    Y = data_object$Y,
    group = data_object$group,
    N = data_object$N,
    G = data_object$G,
    P = data_object$P
  )

  dataset_id <- as.integer(
    str_match(basename(data_file), "_([0-9]+)\\.rds$")[, 2]
  )

  if (method == "SVN") {
    model_file <- svn_model_file
    inits <- NULL
    monitored <- c(
      "delta_lambda", "delta_nu", "lambda", "nu",
      "alpha_f", "psi", "theta"
    )
  } else {
    model_file <- ssp_model_file
    data_jags$pi_a <- 1
    data_jags$pi_b <- ifelse(method == "SSP-W", 1, 4)

    inits <- lapply(seq_len(mcmc_control$n_chains), function(chain_id) {
      make_ssp_inits(
        G = data_object$G,
        P = data_object$P,
        dataset_id = dataset_id,
        chain_id = chain_id
      )
    })

    monitored <- c(
      "delta_lambda", "delta_nu", "lambda", "nu",
      "gamma_lambda", "gamma_nu", "slab_sd",
      "pi_lambda", "pi_nu", "alpha_f", "psi", "theta"
    )
  }

  model <- jags.model(
    file = model_file,
    data = data_jags,
    inits = inits,
    n.chains = mcmc_control$n_chains,
    n.adapt = mcmc_control$n_adapt,
    quiet = TRUE
  )

  update(model, n.iter = mcmc_control$n_burnin, progress.bar = "none")

  )

6.1 收敛诊断

# 汇总多链样本的R-hat与有效样本量
diagnose_fit <- function(samples) {
  variable_names <- varnames(samples)
  ess <- effectiveSize(samples)

  map_dfr(variable_names, function(variable) {
    one_parameter <- samples[, variable]
    all_values <- as.numeric(unlist(one_parameter))

    rhat <- if (!any(is.finite(all_values)) || sd(all_values) == 0) {
      NA_real_
    } else {
      tryCatch(
        gelman.diag(one_parameter, autoburnin = FALSE)$psrf[1, 1],
        error = function(e) NA_real_
      )
    }

    tibble(
      parameter = variable,
      rhat = rhat,
      ess = unname(ess[variable])
    )
  })
}

is_converged <- function(diagnostics, cutoff = 1.10) {
  evaluable <- diagnostics %>% filter(is.finite(rhat))
  nrow(evaluable) > 0L && all(evaluable$rhat < cutoff)
}

6.2 提取载荷和截距结果

get_column <- function(draws, name) {
  if (name %in% colnames(draws)) draws[, name] else rep(NA_real_, nrow(draws))
}

ci95 <- function(x) {
  if (!any(is.finite(x))) return(c(NA_real_, NA_real_))
  unname(quantile(x, c(0.025, 0.975), na.rm = TRUE))
}

# 将JAGS参数名还原为“组×题目”，并与真实值逐项配对
extract_measurement_results <- function(samples, truth, method,
                                        pip_threshold) {
  draws <- as.matrix(samples)
  G <- nrow(truth$lambda)
  P <- ncol(truth$lambda)
  reference_group <- G

  map_dfr(c("loading", "intercept"), function(kind) {
    map_dfr(seq_len(G), function(g) {
      map_dfr(seq_len(P), function(p) {
        fixed <- kind == "loading" && p == 1L
        eligible <- !fixed && g != reference_group

        parameter_name <- if (kind == "loading") "lambda" else "nu"
        difference_name <- if (kind == "loading") "delta_lambda" else "delta_nu"
        gamma_name <- if (kind == "loading") "gamma_lambda" else "gamma_nu"

        posterior <- get_column(
          draws,
          sprintf("%s[%d,%d]", parameter_name, g, p)
        )
        difference <- get_column(
          draws,
          sprintf("%s[%d,%d]", difference_name, g, p)
        )

        true_value <- if (kind == "loading") {
          truth$lambda[g, p]
        } else {
          truth$nu[g, p]
        }
        reference_value <- if (kind == "loading") {
          truth$lambda[reference_group, p]
        } else {
          truth$nu[reference_group, p]
        }

        parameter_ci <- ci95(posterior)
        difference_ci <- ci95(difference)

        pip <- if (method == "SVN" || !eligible) {
          NA_real_
        } else {
          mean(get_column(
            draws,
            sprintf("%s[%d,%d]", gamma_name, g, p)
          ), na.rm = TRUE)
        }

        detected <- if (!eligible) {
          NA_integer_
        } else if (method == "SVN") {
          as.integer(!(difference_ci[1] <= 0 && difference_ci[2] >= 0))
        } else {
          as.integer(pip >= pip_threshold)
        }

        tibble(
          kind = kind,
          group = g,
          item = p,
          fixed = fixed,
          eligible = eligible,
          true_value = true_value,
          true_noninvariant = eligible && abs(true_value - reference_value) > 1e-10,
          estimate = if (fixed) NA_real_ else mean(posterior, na.rm = TRUE),
          posterior_sd = if (fixed) NA_real_ else sd(posterior, na.rm = TRUE),
          ci_lower = parameter_ci[1],
          ci_upper = parameter_ci[2],
          pip = pip,
          detected = detected
        )
      })
    })
  })
}

6.4 计算检验力、第一类错误率与参数返真性指标

# 在载荷和截距内分别计算检测表现与参数返真性指标
summarise_performance <- function(parameter_results) {
  parameter_results %>%
    filter(eligible) %>%
    mutate(
      bias = estimate - true_value,
      squared_error = bias^2,
      coverage = as.integer(
        ci_lower <= true_value & ci_upper >= true_value
      ),
      power_flag = ifelse(true_noninvariant, detected, NA_integer_),
      type1_flag = ifelse(!true_noninvariant, detected, NA_integer_)
    ) %>%
    group_by(kind) %>%
    summarise(
      bias = mean(bias, na.rm = TRUE),
      rmse = sqrt(mean(squared_error, na.rm = TRUE)),
      average_posterior_sd = mean(posterior_sd, na.rm = TRUE),
      empirical_sd = sd(estimate, na.rm = TRUE),
      se_ratio = average_posterior_sd / empirical_sd,
      coverage = mean(coverage, na.rm = TRUE),
      power = mean(power_flag, na.rm = TRUE),
      type1_error = mean(type1_flag, na.rm = TRUE),
      .groups = "drop"
    )
}

6.4 分析一个条件

# 读取与模型对应的最终结果：SVN、SSP-W、SSP-S分别使用三个目录
analyse_condition <- function(condition_dir, method) {
  result_name <- switch(
    method,
    "SVN" = "results_fixed",
    "SSP-W" = "results",
    "SSP-S" = "results_pi2"
  )

  result_files <- list.files(
    file.path(condition_dir, result_name),
    pattern = "^jags_result_[0-9]+\\.rds$",
    full.names = TRUE
  )

  map_dfr(result_files, function(result_file) {
    simulation_id <- as.integer(
      str_match(basename(result_file), "_([0-9]+)\\.rds$")[, 2]
    )

    samples <- readRDS(result_file)
    diagnostics <- diagnose_fit(samples)
    converged <- is_converged(diagnostics, rhat_cutoff)

    data_object <- readRDS(file.path(
      condition_dir,
      "data",
      paste0("data_sim_", simulation_id, ".rds")
    ))

    if (!converged) {
      return(tibble(simulation = simulation_id, converged = FALSE))
    }

    threshold <- if (method == "SVN") NA_real_ else pip_cutoff[[method]]

    extract_measurement_results(
      samples = samples,
      truth = data_object$true_par,
      method = method,
      pip_threshold = threshold
    ) %>%
      mutate(simulation = simulation_id, converged = TRUE)
  })
}

9.1 数据清理与各国等量抽样

library(dplyr)
library(tidyr)
library(readr)

set.seed(20260102)

# 六个移民态度题目；反向计分后，得分越高表示越排斥移民
items <- c(
  "imsmetn", "imdfetn", "impcntr",
  "aljewlv", "almuslv", "algyplv"
)

ess_raw <- read_csv("ESS7e02_3.csv", show_col_types = FALSE)

ess_clean <- ess_raw %>%
  select(cntry, essround, all_of(items)) %>%
  mutate(
    # ESS编码7、8、9分别表示拒答、不知道和无回答
    across(all_of(items), ~ if_else(.x %in% c(7, 8, 9), NA_real_, as.numeric(.x))),
    across(all_of(items), ~ 5 - .x)
  )

# 仅保留六题均有有效作答的被试
ess_complete <- ess_clean %>%
  filter(complete.cases(across(all_of(items))))

# 检查各国各题是否存在极低变异
item_variance <- ess_complete %>%
  pivot_longer(all_of(items), names_to = "item", values_to = "value") %>%
  group_by(cntry, item) %>%
  summarise(variance = var(value), .groups = "drop")

stopifnot(!any(is.na(item_variance$variance)))

# 只保留有效样本不少于500人的国家，并从每国随机抽取500人
country_n <- ess_complete %>% count(cntry, name = "N")

ess_N500 <- ess_complete %>%
  semi_join(country_n %>% filter(N >= 500), by = "cntry") %>%
  group_by(cntry) %>%
  slice_sample(n = 500, replace = FALSE) %>%
  ungroup()

# 每个国家必须恰好保留500名被试
stopifnot(all(count(ess_N500, cntry)$n == 500))
write_csv(ess_N500, "ESS7_N500.csv")

9.3 拟合SVN、SSP-W与SSP-S

实证研究沿用主模拟中的三种模型。bsem_model.jags、model.jags和model_pi2.jags分别对应SVN、SSP-W和SSP-S；后两者的模型结构相同，仅包含概率先验分别为Beta(1, 1)和Beta(1, 4)。下列迭代次数与实证分析的最终运行设置一致。

fit_empirical_model <- function(method, data_jags) {
  control <- empirical_control[[method]]

  if (method == "SVN") {
    inits <- NULL
    monitored <- c(
      "delta_lambda", "delta_nu", "lambda", "nu",
      "alpha_f", "psi", "theta"
    )
  } else {
    # 沿用第5.1节的SSP初始化函数，为四条链设置不同随机种子
    inits <- lapply(1:4, function(chain_id) {
      make_ssp_inits(
        G = data_jags$G,
        P = data_jags$P,
        dataset_id = 1L,
        chain_id = chain_id
      )
    })
    monitored <- c(
      "delta_lambda", "delta_nu", "lambda", "nu",
      "gamma_lambda", "gamma_nu", "alpha_f", "psi", "theta",
      "slab_sd", "pi_lambda", "pi_nu"
    )
  }

  model <- jags.model(
    file = control$model_file,
    data = data_jags,
    inits = inits,
    n.chains = 4,
    n.adapt = 2000,
    quiet = TRUE
  )

  update(model, n.iter = control$burnin, progress.bar = "none")

  samples <- coda.samples(
    model = model,
    variable.names = monitored,
    n.iter = control$sample,
    thin = 1,
    progress.bar = "none"
  )

  # 保存完整后验样本、收敛指标和后验摘要
  dir.create(control$result_dir, showWarnings = FALSE, recursive = TRUE)
  saveRDS(
    samples,
    file.path(control$result_dir, "ESS_ref_DE_samples.rds")
  )

  fit_summary <- list(
    rhat = gelman.diag(samples, autoburnin = FALSE)$psrf[, 1],
    ess = effectiveSize(samples),
    posterior = summary(samples)
  )
  saveRDS(
    fit_summary,
    file.path(control$result_dir, "ESS_ref_DE_summary.rds")
  )

  invisible(samples)
}

fit_empirical_model("SVN", data_jags)
fit_empirical_model("SSP-W", data_jags)
fit_empirical_model("SSP-S", data_jags)
